=== .github/workflows/ci.yml ===
name: CI

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v3

    - name: Setup Deno
      uses: denoland/setup-deno@v1
      with:
        deno-version: v1.x

    - name: Setup Node.js
      uses: actions/setup-node@v3
      with:
        node-version: '18'

    - name: Verify formatting
      run: deno fmt --check

    - name: Lint
      run: deno lint

    - name: Type check
      run: deno check supabase/functions/**/*.ts

    - name: Run tests
      run: deno test --allow-all --no-check
